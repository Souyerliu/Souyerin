        ```r
        A <- matrix(c(3, 1, -1,
                5, 2, 4), nrow = 2, byrow = TRUE)
        t(A) # 转置
        B <- matrix(c(1, 2, -2,
                    5, -3, 1), nrow = 2, byrow = TRUE)
        4 * A # 矩阵标量乘法
        A + B # 矩阵加法
        A %*% t(B) # 矩阵乘法 

        C <- c(-2,7,9)

        A %*% C # 矩阵向量乘法（自动确定行/列向量）

        D <- matrix(c(3, 4,
                    2, 1), nrow = 2, byrow = TRUE)
        solve(D) # 矩阵求逆

        F <- matrix(c(3/2, -1/2, 0,
                    -1/2, 3/2, 0,
                    0, 0, 1), nrow = 3, byrow = TRUE)

        F_eigen <- eigen(F)
        F_value <- F_eigen$values # 矩阵特征值（向量形式）
        F_vector <- F_eigen$vectors # 矩阵特征向量（矩阵形式）

        library(psych)
        tr(F) # 矩阵的迹
        det(F) # 矩阵行列式
        ```

# 绘制椭圆  
+ 旋转椭圆的二次型形式为
    $$
    \begin{bmatrix}
    x \\
    y
    \end{bmatrix}
    A_1
    \begin{bmatrix}
    x & y
    \end{bmatrix}=\begin{bmatrix}
    x \\
    y
    \end{bmatrix}
    \begin{bmatrix}
    a_1 & \frac{a_3}{2} \\
    \frac{a_3}{2} & a_2
    \end{bmatrix}
    \begin{bmatrix}
    x & y
    \end{bmatrix}
    = a_1x^2 + a_2y^2 + a_3xy = 1
    $$

        ```r
        library(mvtnorm)
        # 多元正态分布的均值
        ```

# 正态性检验
1. Q-Q plot绘图检验
    - 将样本$x_1,x_2,\dots,x_n$从小到大排序得到$x_{(1)},x_{(2)},\dots,x_{(n)}$，它们各自对应的概率值为$\displaystyle\frac{1-\frac{1}{2}}{n},\frac{2-\frac{1}{2}}{n},\dots,\frac{n-\frac{1}{2}}{n}$。
    - 计算标准正态分布的下分位数$q_{(1)},q_{(2)},\dots,q_{(n)}$。
    - 画出$(q_{(1)},x_{(1)}),(q_{(2)},x_{(2)}),\dots,(q_{(n)},x_{(n)})$的图，并检查这些点是否都近似地在一条直线上。

    当然也可以直接用`qqnorm()`函数，此时概率值变为$\displaystyle \frac{1-\frac{3}{8}}{n-\frac{1}{4}},\frac{2-\frac{3}{8}}{n-\frac{1}{4}},\dots,\frac{n-\frac{3}{8}}{n-\frac{1}{4}}$。
    + 同时也可以用`qqline()`绘制参考线（基于样本四分位数与正态分布四分位数点连线得到）
    + 更严谨的方法是计算Q-Q plot的相关系数，可参考下述检验表格：
        | n | .01 | .05 | .10 |
        | :---: | :---: | :---: | :---: |
        | 5 | .8299 | .8788 | .9032 |
        | 10 | .8801 | .9198 | .9351 |
        | 15 | .9126 | .9389 | .9503 |

        当指定样本数与显著性水平下相关系数小于临界值，则拒绝正态性假设。

        ```python
        import numpy as np
        import matplotlib.pyplot as plt
        from scipy import stats

        x = np.array([-1, -0.1, 0.16, 0.41, 0.62, 0.8, 1.26, 1.54, 1.71, 2.3])
        n = len(x)

        prob_qqnorm = (np.arange(1, n + 1) - 3/8) / (n + 1/4)
        q_qqnorm = stats.norm.ppf(prob_qqnorm)
        x_sorted = np.sort(x)

        rank = stats.rankdata(x)
        prob_values = (rank - 0.5) / n
        q = stats.norm.ppf(prob_values)

        fig, ax = plt.subplots(figsize=(6, 6))

        ax.scatter(q_qqnorm, x_sorted, color='black', label='qqnorm()', zorder=3)
        ax.scatter(q, x, color='blue', label='custom (rank-1/2)/n', zorder=3)

        probs = np.array([0.25, 0.75])
        y_q = np.quantile(x, probs, method='linear')
        x_q = stats.norm.ppf(probs)
        slope = (y_q[1] - y_q[0]) / (x_q[1] - x_q[0])
        intercept = y_q[0] - slope * x_q[0]

        xx = np.linspace(q_qqnorm.min(), q_qqnorm.max(), 100)
        ax.plot(xx, intercept + slope * xx, color='red', label='qqline')

        ax.set_xlabel('Theoretical Quantiles')
        ax.set_ylabel('Sample Quantiles')
        ax.set_title('Normal Q-Q Plot')
        ax.legend()
        plt.show()

        r = np.corrcoef(x_sorted, q_qqnorm)[0, 1]
        print(r)
        ```

2. 多变量正态性检验  
    使用Chi-square plot绘图检验。具体方法：计算马氏距离平方
    $$
    d_j^2=(\mathbf{x}_j-\overline{\mathbf{x}})^\top\mathbf{S}^{-1}(\mathbf{x}_j-\overline{\mathbf{x}}),\quad j=1,2,\dots,n
    $$
    并把它们从小到大排序得到$d_{(1)}^2,d_{(2)}^2,\dots,d_{(n)}^2$。然后画出
    $$
    \left(q_{c,p}\left(\frac{j-\frac{1}{2}}{n}\right),d_{(j)}^2\right)
    $$
    这里$\displaystyle q_{c,p}\left(\frac{j-\frac{1}{2}}{n}\right)$是$\chi_p^2$的$100\cdot\dfrac{j-\frac{1}{2}}{n}$下分位数。
    <Tabs class="code-group" defaultValue="r">   
        <Tab label="R" value="r">
        ```r
        # 读入数据
        x1 <- c(108.28, 152.36, 95.04, 65.45, 62.97, 263.99, 265.19, 285.06, 92.01, 165.68)
        x2 <- c(17.05, 16.59, 10.91, 14.14, 9.52, 25.33, 18.54, 15.73, 8.10, 11.13)
        X <- cbind(x1, x2)
        # 计算 d_j^2
        d2 <- mahalanobis(X,
                        center = colMeans(X), # 样本的均值向量
                        cov = cov(X)) # 样本的协方差矩阵
        # 1. 排序（从小到大）
        d2.sorted <- sort(d2)
        # 2. 绘图
        d2.q <- qchisq((rank(d2.sorted) - 1/2) / length(d2.sorted),
                    ncol(X))
        plot(d2.q, d2.sorted, pch = 19, col = "blue")

        library(MVN)
        library(ggplot2)

        p <- multivariate_diagnostic_plot(X, type = "qq")

        # 在返回的 ggplot 对象上叠加散点
        p + geom_point(aes(x = d2.sorted, y = d2.q), 
                    color = "blue",
        ```

        ```python
        import numpy as np
        from scipy.optimize import minimize
        from scipy import stats

        x = np.array([
            0.15, 0.09, 0.18, 0.10, 0.05, 0.12, 0.08, 0.05,
            0.08, 0.10, 0.07, 0.02, 0.01, 0.10, 0.10, 0.10,
            0.02, 0.10, 0.01, 0.40, 0.10, 0.05, 0.03, 0.05,
            0.15, 0.10, 0.15, 0.09, 0.08, 0.18, 0.10, 0.20,
            0.11, 0.30, 0.02, 0.20, 0.20, 0.30, 0.30, 0.40,
            0.30, 0.05
        ])

        x1 = x.copy()
        x2 = np.array([
            0.30, 0.09, 0.30, 0.10, 0.10, 0.12, 0.09, 0.10, 0.09, 0.10,
            0.07, 0.05, 0.01, 0.45, 0.12, 0.20, 0.04, 0.10, 0.01, 0.60,
            0.12, 0.10, 0.05, 0.05, 0.15, 0.30, 0.15, 0.09, 0.09, 0.28,
            0.10, 0.10, 0.10, 0.30, 0.12, 0.25, 0.20, 0.40, 0.33, 0.32,
            0.12, 0.12
        ])

        X = np.column_stack((x1, x2))

        def boxcox_transform(x, lam):
            if np.isclose(lam, 0):
                return np.log(x)
            return (x**lam - 1) / lam

        def neg_log_likelihood_multivariate(lambdas, X):
            n, p = X.shape
            Y = np.empty_like(X, dtype=float)
            log_jac = 0.0
            for j in range(p):
                lam = lambdas[j]
                Y[:, j] = boxcox_transform(X[:, j], lam)
                log_jac += (lam - 1) * np.sum(np.log(X[:, j]))
            Y_centered = Y - Y.mean(axis=0)
            Sigma = (Y_centered.T @ Y_centered) / n
            sign, logdet = np.linalg.slogdet(Sigma)
            nll = (n / 2) * logdet - log_jac
            return nll

        lam1_init = stats.boxcox_normmax(x1, method="mle")
        lam2_init = stats.boxcox_normmax(x2, method="mle")
        res = minimize(neg_log_likelihood_multivariate, [lam1_init, lam2_init], args=(X,), method='Nelder-Mead')
        print("Joint lambdas:", res.x)

        print("Marginal lambda x1:", stats.boxcox_normmax(x1, method="mle"))
        print("Marginal lambda x2:", stats.boxcox_normmax(x2, method="mle"))
        ```
